# Probe v3: fully SYNCHRONOUS. Self-terminates via the 8th MXCFB_SEND_UPDATE
# ioctl (clean quit => flushed log). No async, no interrupt, no detach.
set architecture arm
set pagination off
set confirm off
set breakpoint pending on
set print pretty off
file /home/user/kobo-emu/build/rootfs-4.38.23684/usr/local/Kobo/nickel
set sysroot /home/user/kobo-emu/build/rootfs-4.38.23684
target remote :1235
# OpenSSL ARMcaps probe deliberately executes possibly-unsupported insns
# and catches the SIGILL itself; let it pass or the session stops in
# _armv7_tick during every library load.
handle SIGILL nostop noprint pass
set $c_main = 0
set $c_exec = 0
set $c_beginpaint = 0
set $c_resize = 0
set $c_setdirty = 0
set $c_blit = 0
set $c_blitv2 = 0
set $c_doredraw = 0
set $c_really = 0
set $c_dispflush = 0
set $c_ionmap = 0
set $c_qbsflush = 0
set $c_su = 0

break main
commands
silent
set $c_main = $c_main + 1
if $c_main <= 2
printf "\n>>> HIT main #%d\n", $c_main
bt 4
end
continue
end

break _ZN16QCoreApplication4execEv
commands
silent
set $c_exec = $c_exec + 1
if $c_exec <= 2
printf "\n>>> HIT QCoreApplication::exec #%d\n", $c_exec
bt 4
end
continue
end

break _ZN27KoboFrameBufferBackingStore10beginPaintERK7QRegion

silent
set $c_beginpaint = $c_beginpaint + 1
if $c_beginpaint <= 5
printf "\n>>> HIT KoboFrameBufferBackingStore::beginPaint #%d this=", $c_beginpaint
p/x $r0
bt 6
end
continue
end

break _ZN27KoboFrameBufferBackingStore6resizeERK5QSizeRK7QRegion

silent
set $c_resize = $c_resize + 1
if $c_resize <= 5
printf "\n>>> HIT KoboFrameBufferBackingStore::resize #%d size=", $c_resize
x/2dw $r1

end

end

break _ZN13KoboScreenNXP8setDirtyERK5QRectj

printf "\n>>> HIT KoboScreenNXP::setDirty #%d rect=", $c_setdirty

printf "flags="

bt 5
end

end

break _ZN13KoboScreenNXP4blitERK7QRegion

silent
set $c_blit = $c_blit + 1
if $c_blit <= 8
printf "\n>>> HIT KoboScreenNXP::blit #%d\n", $c_blit

end

end

break _ZN13KoboScreenNXP6blitV2ERK7QRegion

break _ZN13KoboScreenNXP14reallyDoRedrawEv

printf "\n>>> HIT KoboScreenNXP::reallyDoRedraw #%d\n", $c_really

end

break _ZN4Disp5flushERK5QRectim
commands
silent
set $c_dispflush = $c_dispflush + 1
if $c_dispflush <= 8
printf "\n>>> HIT Disp::flush #%d rect=", $c_dispflush
x/4dw $r1
printf "arg2="

break _ZN3Ion3mapEv
commands
silent
set $c_ionmap = $c_ionmap + 1

printf "\n>>> HIT Ion::map #%d\n", $c_ionmap

break _ZN13QBackingStore5flushERK7QRegionP7QWindowRK6QPointRK5QListI5QPairI5QRectjEE

# MXCFB_SEND_UPDATE trap: ioctl nr 0x2e, type 'F' (0x46). Terminates the probe.
break ioctl if ((($r1 >> 8) & 255) == 70) && (($r1 & 255) == 46)
commands
silent
set $c_su = $c_su + 1
if $c_su <= 16
printf "\n>>> HIT MXCFB_SEND_UPDATE ioctl #%d req=", $c_su
p/x $r1
printf "update_data="
x/8xw $r2
bt 6
end
if $c_su == 16
printf "\n=== 16 SEND_UPDATEs seen; final counters ===\n"
p $c_main
p $c_exec
p $c_beginpaint
p $c_resize
p $c_setdirty
p $c_blit
p $c_blitv2
p $c_doredraw
p $c_really
p $c_dispflush
p $c_ionmap
p $c_qbsflush
p $c_su
printf "\n=== all-thread backtraces ===\n"
thread apply all bt 10
quit
end
continue
end

printf "\n=== planted, synchronous continue ===\n"
continue
printf "\n=== continue returned (target exited?) ===\n"
p $c_main
p $c_exec
p $c_beginpaint
p $c_resize
p $c_setdirty
p $c_blit
p $c_blitv2
p $c_doredraw
p $c_really
p $c_dispflush
p $c_ionmap
p $c_qbsflush
p $c_su
quit
