# Phase 2.4 gdb probe: paint-path breakpoints + timed interrupt for idle stacks.
# Run: gdb -x /tmp/opencode/gdb-probe.txt (NOT --batch: pending breakpoints
# must auto-answer; 'set breakpoint pending on' handles that).
set architecture arm
set pagination off
set confirm off
set breakpoint pending on
set print pretty off
file /home/user/kobo-emu/build/rootfs-4.38.23684/usr/local/Kobo/nickel
set sysroot /home/user/kobo-emu/build/rootfs-4.38.23684

set $c_resize = 0

set $c_blitv2 = 0
set $c_doredraw = 0

printf "\n>>> HIT KoboFrameBufferBackingStore::beginPaint #%d\n", $c_beginpaint

end
