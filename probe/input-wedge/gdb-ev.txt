# Probe v2, session 1: event logger. Plants pending breakpoints, runs async,
# sleeps 150s INSIDE gdb, then DETACHES (guest keeps running for session 2)
# and quits cleanly (output flushed).
set architecture arm
set pagination off
set confirm off
set breakpoint pending on
set print pretty off
file /home/user/kobo-emu/build/rootfs-4.38.23684/usr/local/Kobo/nickel
set sysroot /home/user/kobo-emu/build/rootfs-4.38.23684

set $c_exec = 0
set $c_beginpaint = 0

set $c_ionmap = 0

break main
commands
silent
set $c_main = $c_main + 1
if $c_main <= 2
printf "\n>>> HIT main #%d\n", $c_main
bt 4
end
continue
end
